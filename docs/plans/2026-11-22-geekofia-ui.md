# geekofia_ui Gem Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A public Rails engine gem, `geekofia_ui`, that gives ERB + Tailwind v4 apps shadcn-style components as `ui_*` helpers, a form builder, two Stimulus controllers (clipboard, theme), a light/dark theme with a toggle, and pluggable icon and font providers.

**Architecture:** A non-isolated Rails engine. Helpers live in `lib/geekofia_ui/helpers/*.rb` (explicitly required) and are included into `ActionView::Base` on load; atomic components use tag helpers, structural ones render partials in `app/views/geekofia_ui/components/`. Theme tokens ship in `app/assets/tailwind/geekofia_ui/engine.css`, which tailwindcss-rails ≥ 4.2 builds into the host's `app/assets/builds/tailwind/geekofia_ui.css`. Stimulus controllers are pinned through the engine's own `config/importmap.rb`. Icon and font providers are small Ruby objects selected by `GeekofiaUi.configure`.

**Tech Stack:** Ruby 3.3, Rails 8.1 engine, Tailwind CSS v4 (tailwindcss-rails 4.7), importmap-rails, Stimulus, Minitest, Lucide SVGs (ISC).

Spec: `server-wizard/docs/superpowers/specs/2026-11-22-geekofia-ui-and-revamp-design.md` (Part 1).

```ruby
module GeekofiaUi
  # Joins Tailwind class lists. When two utilities belong to the same group
  # (and share the same variant prefix), the later one wins.
  module ClassMerge
    GROUPS = [
      [ :px, /\Apx-/ ], [ :py, /\Apy-/ ], [ :pt, /\Apt-/ ], [ :pr, /\Apr-/ ], [ :pb, /\Apb-/ ], [ :pl, /\Apl-/ ], [ :p, /\Ap-/ ],
      [ :mx, /\A-?mx-/ ], [ :my, /\A-?my-/ ], [ :mt, /\A-?mt-/ ], [ :mr, /\A-?mr-/ ], [ :mb, /\A-?mb-/ ], [ :ml, /\A-?ml-/ ], [ :m, /\A-?m-/ ],
      [ :size, /\Asize-/ ], [ :min_w, /\Amin-w-/ ], [ :max_w, /\Amax-w-/ ], [ :w, /\Aw-/ ], [ :h, /\Ah-/ ],
      [ :text_size, /\Atext-(xs|sm|base|lg|[2-9]?xl)\z/ ], [ :text_align, /\Atext-(left|center|right|justify|start|end)\z/ ],
      [ :text_color, /\Atext-/ ],
      [ :font_weight, /\Afont-(thin|extralight|light|normal|medium|semibold|bold|extrabold|black)\z/ ],
      [ :font_family, /\Afont-(sans|serif|mono)\z/ ],
      [ :bg, /\Abg-/ ], [ :border_width, /\Aborder(-[xytrbl])?(-\d+)?\z/ ], [ :border_color, /\Aborder-/ ],
      [ :rounded, /\Arounded(-|\z)/ ], [ :gap, /\Agap-/ ], [ :shadow, /\Ashadow(-|\z)/ ], [ :opacity, /\Aopacity-/ ],
      [ :justify, /\Ajustify-/ ], [ :items, /\Aitems-/ ],
      [ :display, /\A(block|inline-block|inline|flex|inline-flex|grid|inline-grid|hidden|contents|table)\z/ ]
    ].freeze

    module_function

    def call(*lists)
      tokens = lists.flatten.compact.flat_map { |list| list.to_s.split }
      result = []
      positions = {}
      tokens.each do |token|
        key = group_key(token)
        if key && positions.key?(key)
          result[positions[key]] = nil
        end
        positions[key] = result.size if key
        result << token
      end
      result.compact.uniq.join(" ")
    end

    def group_key(token)
      variants, _, utility = token.rpartition(":")
      utility = utility.delete_prefix("!")
      group = GROUPS.find { |_, pattern| utility.match?(pattern) }&.first
      "#{variants}:#{group}" if group
    end
  end
end
```

```ruby
      copy: "copy", check: "check", sun: "sun", moon: "moon", monitor: "monitor", info: "info",
      alert_triangle: "triangle-alert", alert_circle: "circle-alert", check_circle: "circle-check", x: "x",
      plus: "plus", server: "server", terminal: "terminal", refresh: "refresh-cw", pencil: "pencil",

      log_out: "log-out", hard_drive: "hard-drive", key: "key-round", play: "play", plug: "plug"
    }.freeze

    BUILT_IN = { lucide: "Lucide", iconify: "Iconify", font_awesome: "FontAwesome", material_symbols: "MaterialSymbols" }.freeze

    def self.resolve(provider, config)
      return provider unless provider.is_a?(Symbol) || provider.is_a?(String)

        raise ArgumentError, "unknown icon_provider #{provider.inspect}; use one of #{BUILT_IN.keys.join(", ")} or an object"
      end
      const_get(class_name).new(config)
    end
  end
end
```

```ruby
module GeekofiaUi
  module Icons
    class Base
      def initialize(config)
        @config = config
      end

      def head_tags = "".html_safe

      private

      # Alias first, then the provider's semantic map, otherwise the name as given.
      def provider_name(name, map)
        key = name.to_s.to_sym
        aliased = @config.icon_aliases[key]
        return aliased.to_s if aliased
        map.fetch(key, name.to_s)
      end

      def element(tag, attributes, inner = "")
        attrs = attributes.compact.map { |k, v| %(#{k}="#{ERB::Util.html_escape(v)}") }.join(" ")
        "<#{tag} #{attrs}>#{inner}</#{tag}>".html_safe
      end

      def classes(*parts) = parts.compact.join(" ")
    end
  end
end
```

```ruby
module GeekofiaUi
  module Icons
    class Lucide < Base
      DIR = File.expand_path("lucide", __dir__)
      NAME = /\A[a-z0-9-]+\z/

      def self.inner_svg(file)
        @cache ||= {}
        @cache.fetch(file) do
          path = File.join(DIR, "#{file}.svg")
          @cache[file] = File.exist?(path) ? File.read(path)[%r{<svg[^>]*>(.*)</svg>}m, 1]&.strip : nil
        end
      end

      def render(name, size:, css_class: nil, attrs: {})
        file = provider_name(name, LUCIDE_NAMES)
        inner = self.class.inner_svg(file) if file.match?(NAME)
        return missing(name) unless inner

        element(:svg, {
          xmlns: "http://www.w3.org/2000/svg", width: size, height: size, viewBox: "0 0 24 24",
          fill: "none", stroke: "currentColor", "stroke-width": 2, "stroke-linecap": "round",
          "stroke-linejoin": "round", class: classes("shrink-0", css_class)
        }.merge(attrs), inner)
      end

      private

      def missing(name)
        raise ArgumentError, "unknown lucide icon #{name.inspect}" if !defined?(Rails) || Rails.env.local?
        "".html_safe
      end
    end
  end
end
```

```ruby
module GeekofiaUi
  module Icons
    class Iconify < Base
      SCRIPT = "https://code.iconify.design/iconify-icon/2.3.0/iconify-icon.min.js"

      def render(name, size:, css_class: nil, attrs: {})
        element(:"iconify-icon", { icon: icon_name(name), width: size, height: size,
                                   class: classes("shrink-0", css_class) }.merge(attrs))
      end

      def head_tags = %(<script src="#{SCRIPT}"></script>).html_safe

      private

      def icon_name(name)
        key = name.to_s.to_sym
        return qualify(@config.icon_aliases[key].to_s) if @config.icon_aliases[key]
        return "lucide:#{LUCIDE_NAMES[key]}" if LUCIDE_NAMES.key?(key)
        qualify(name.to_s)
      end

      def qualify(name) = name.include?(":") ? name : "#{@config.icon_set}:#{name}"
    end
  end
end
```

```ruby
require "erb"
require "active_support/core_ext/string/output_safety"

module GeekofiaUi
  module Fonts
    SANS_STACK = 'ui-sans-serif, system-ui, -apple-system, "Segoe UI", Roboto, "Helvetica Neue", Arial, sans-serif'
    MONO_STACK = 'ui-monospace, SFMono-Regular, Menlo, Monaco, Consolas, "Liberation Mono", monospace'
    BUILT_IN = { system: "System", google: "Google", bunny: "Bunny", self_hosted: "SelfHosted" }.freeze

    def self.resolve(provider, config)
      return provider unless provider.is_a?(Symbol) || provider.is_a?(String)
      class_name = BUILT_IN.fetch(provider.to_sym) do
        raise ArgumentError, "unknown font_provider #{provider.inspect}; use one of #{BUILT_IN.keys.join(", ")} or an object"
      end
      const_get(class_name).new(config)
    end

    class Base
      def initialize(config)
        @config = config
      end

      def head_tags = "".html_safe

      def stacks
        { sans: stack(@config.font_sans, SANS_STACK), mono: stack(@config.font_mono, MONO_STACK) }
      end

      private

      def stack(family, fallback)
        family.to_s.strip.empty? ? fallback : %("#{family.to_s.delete('"<>')}", #{fallback})
      end

      def families = [ @config.font_sans, @config.font_mono ].map(&:to_s).map(&:strip).reject(&:empty?).uniq

      def link(href) = %(<link rel="stylesheet" href="#{ERB::Util.html_escape(href)}">)
      def preconnect(href, crossorigin: false) = %(<link rel="preconnect" href="#{href}"#{" crossorigin" if crossorigin}>)
    end

    class System < Base
      def stacks = { sans: SANS_STACK, mono: MONO_STACK }
    end

    class SelfHosted < Base
    end

    class Google < Base
      def head_tags
        return "".html_safe if families.empty?
        query = families.map { |f| "family=#{f.tr(" ", "+")}:wght@#{@config.font_weights.join(";")}" }.join("&")
        [ preconnect("https://fonts.googleapis.com"), preconnect("https://fonts.gstatic.com", crossorigin: true),
          link("https://fonts.googleapis.com/css2?#{query}&display=swap") ].join("\n").html_safe
      end
    end

    class Bunny < Base
      def head_tags
        return "".html_safe if families.empty?
        query = families.map { |f| "#{f.downcase.tr(" ", "-")}:#{@config.font_weights.join(",")}" }.join("|")
        [ preconnect("https://fonts.bunny.net"), link("https://fonts.bunny.net/css?family=#{query}&display=swap") ].join("\n").html_safe
      end
    end
  end
end
```

```js
import { Controller } from "@hotwired/stimulus"

// Copies textValue, or the textContent of the element matched by sourceValue,
// then sets data-copied="true" on the button for a moment.
export default class extends Controller {
  static values = { text: String, source: String, duration: { type: Number, default: 2000 } }

  async copy(event) {
    event.preventDefault()
    const text = this.sourceValue ? (document.querySelector(this.sourceValue)?.textContent ?? "") : this.textValue
    try {
      await navigator.clipboard.writeText(text)
    } catch {
      this.fallback(text)
    }
    this.element.dataset.copied = "true"
    clearTimeout(this.timer)
    this.timer = setTimeout(() => { delete this.element.dataset.copied }, this.durationValue)
  }

  fallback(text) {
    const area = document.createElement("textarea")
    area.value = text
    area.setAttribute("readonly", "")
    area.style.position = "fixed"
    area.style.opacity = "0"
    document.body.appendChild(area)
    area.select()
    document.execCommand("copy")
    area.remove()
  }

  disconnect() {
    clearTimeout(this.timer)
  }
}
```

```js
import { Controller } from "@hotwired/stimulus"

const ORDER = ["light", "dark", "system"]

// Cycles light → dark → system. The pre-paint script from ui_head_tags owns
// applying the class; this controller only stores the choice and re-applies.
export default class extends Controller {
  connect() {
    this.render()
  }

  cycle(event) {
    event.preventDefault()
    const next = ORDER[(ORDER.indexOf(this.current) + 1) % ORDER.length]
    try { localStorage.setItem(this.key, next) } catch {}
    window.GeekofiaUI?.applyTheme()
    this.render()
  }

  get key() {
    return window.GeekofiaUI?.storageKey || "geekofia-ui-theme"
  }

  get current() {
    try { return localStorage.getItem(this.key) || "system" } catch { return "system" }
  }

  render() {
    this.element.dataset.theme = this.current
    this.element.title = `Theme: ${this.current}`
  }
}
```

```ruby
require "action_view"

module GeekofiaUi
  # form_with(model:, builder: GeekofiaUi::FormBuilder) { |f| f.ui_text_field :name, hint: "…" }
  class FormBuilder < ActionView::Helpers::FormBuilder
    INPUT = "flex h-9 w-full rounded-md border border-input bg-transparent px-3 py-1 text-sm shadow-xs transition-colors " \
            "placeholder:text-muted-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring " \
            "disabled:cursor-not-allowed disabled:opacity-50 aria-invalid:border-destructive aria-invalid:ring-destructive/30"
    TEXTAREA = INPUT.sub("flex h-9", "flex min-h-20").sub("py-1", "py-2")
    LABEL = "text-sm font-medium leading-none"

    %i[text_field email_field password_field number_field].each do |kind|
      define_method(:"ui_#{kind}") do |method, label: nil, hint: nil, **options|
        ui_field(method, label:, hint:) { |aria| public_send(kind, method, **control_options(options, INPUT, aria)) }
      end
    end

    def ui_text_area(method, label: nil, hint: nil, **options)
      ui_field(method, label:, hint:) { |aria| text_area(method, **control_options(options, TEXTAREA, aria)) }
    end

    def ui_select(method, choices, options = {}, html_options = {})
      options = options.dup
      label = options.delete(:label)
      hint = options.delete(:hint)
      ui_field(method, label:, hint:) { |aria| select(method, choices, options, control_options(html_options, INPUT, aria)) }
    end

    def ui_submit(value = nil, variant: :default, **options)
      @template.ui_button(value || submit_default_value, type: "submit", variant:, **options)
    end

    private

    def ui_field(method, label:, hint:)
      base_id = field_id(method)
      hint_id = "#{base_id}_hint"
      error_id = "#{base_id}_error"
      messages = object.respond_to?(:errors) ? object.errors.full_messages_for(method) : []
      described_by = [ (hint_id if hint), (error_id if messages.any?) ].compact.join(" ").presence
      aria = { invalid: (true if messages.any?), describedby: described_by }.compact

      @template.content_tag(:div, class: "space-y-2", data: { ui_field: method }) do
        @template.safe_join([
          label(method, label, class: LABEL),
          yield(aria),
          (@template.content_tag(:p, hint, id: hint_id, class: "text-xs text-muted-foreground") if hint),
          (@template.content_tag(:p, messages.to_sentence, id: error_id, class: "text-xs font-medium text-destructive") if messages.any?)
        ].compact)
      end
    end

    def control_options(options, base, aria)
      options = options.dup
      options[:class] = GeekofiaUi::ClassMerge.call(base, options[:class])
      options[:aria] = (options[:aria] || {}).merge(aria)
      options
    end
  end
end
```

```ruby
require "rails/generators"

module GeekofiaUi
  module Generators
    class InstallGenerator < Rails::Generators::Base
      source_root File.expand_path("templates", __dir__)
      desc "Wire geekofia_ui into this app: Tailwind import, Stimulus controllers, initializer, head tags."

      CSS_FILE = "app/assets/tailwind/application.css"
      CSS_IMPORT = %(@import "../builds/tailwind/geekofia_ui";)
      JS_FILE = "app/javascript/controllers/index.js"
      JS_LINE = %(eagerLoadControllersFrom("geekofia_ui/controllers", application))
      LAYOUT = "app/views/layouts/application.html.erb"

      def add_tailwind_import
        return skip(CSS_FILE) unless exist?(CSS_FILE)
        return if read(CSS_FILE).include?(CSS_IMPORT)
        inject_into_file CSS_FILE, "#{CSS_IMPORT}\n", after: /@import "tailwindcss";\n/
      end

      def register_controllers
        return skip(JS_FILE) unless exist?(JS_FILE)
        return if read(JS_FILE).include?(JS_LINE)
        append_to_file JS_FILE, "#{JS_LINE}\n"
      end

      def create_initializer
        copy_file "geekofia_ui.rb", "config/initializers/geekofia_ui.rb", skip: true
      end

      def add_head_tags
        return skip(LAYOUT) unless exist?(LAYOUT)
        return if read(LAYOUT).include?("ui_head_tags")
        inject_into_file LAYOUT, "    <%= ui_head_tags %>\n", before: /^\s*<%= stylesheet_link_tag/
      end

      private

      def exist?(path) = File.exist?(File.join(destination_root, path))
      def read(path) = File.read(File.join(destination_root, path))
      def skip(path) = say_status(:skip, "#{path} not found", :yellow)
    end
  end
end
```
